cmake_minimum_required(VERSION 3.21)
project(CyInsightsWindowsCollector LANGUAGES CXX RC)
if(MSVC)
    string(REPLACE "┬á" " " CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${CMAKE_CXX_CL_SHOWINCLUDES_PREFIX}")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CMAKE_CXX_CL_SHOWINCLUDES_PREFIX}")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The ReShade collector requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(CYGI_VENDOR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../ThirdParty/CyGPUInspector" ABSOLUTE)
set(CYGI_OUTPUT_DIR "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/Zc:__cplusplus;/utf-8;/EHsc;/sdl;/guard:cf>")
add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
add_subdirectory("${CYGI_VENDOR_ROOT}/CyGPUInspectorCore" Core)
target_compile_definitions(CyGPUInspectorCore PUBLIC CYINSIGHTS_WINDOWS_GPU=1)
set(CYINSIGHTS_WINDOWS_GPU ON)
add_subdirectory("${CYGI_VENDOR_ROOT}/CyGPUInspectorRS" Collector)
set_target_properties(CyGPUInspectorRS PROPERTIES OUTPUT_NAME "CyInsightsGpu")
add_library(CyInsightsWindowsGpuCollector ALIAS CyGPUInspectorRS)
install(TARGETS CyGPUInspectorRS RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${CYGI_VENDOR_ROOT}/CyGPUInspectorRS/LICENSE" DESTINATION . RENAME LICENSE-CyInsightsCollector-MIT.txt)
install(FILES "${CYGI_VENDOR_ROOT}/ThirdParty/reshade/LICENSE.md" DESTINATION LICENCES/ReShade-SDK)
install(FILES "${CYGI_VENDOR_ROOT}/ThirdParty/reshade_imgui/LICENSE.txt" DESTINATION LICENCES/DearImGui-ReShade)
