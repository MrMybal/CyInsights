cmake_minimum_required(VERSION 3.21)
project(CyInsightOVR LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IMGUI "${ROOT}/ThirdParty/CyGPUInspector/ThirdParty/imgui")
add_library(OvrImGui STATIC "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp" "${ROOT}/ThirdParty/ImGuiOpenGL/imgui_impl_opengl3.cpp")
set_target_properties(OvrImGui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(OvrImGui PUBLIC "${IMGUI}" "${ROOT}/ThirdParty/ImGuiOpenGL")
target_compile_definitions(OvrImGui PRIVATE IMGUI_IMPL_OPENGL_ES3)
set(CYUI_IMGUI_TARGET OvrImGui CACHE STRING "" FORCE)
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/ThirdParty/CyUI" "${CMAKE_BINARY_DIR}/cyui")
set_target_properties(cyui PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(cyinsightovr SHARED native/Main.cpp native/OpenXrCapabilities.cpp "${ROOT}/ThirdParty/CyGPUInspector/CyGPUInspectorCore/Source/Json.cpp")
target_include_directories(cyinsightovr PRIVATE "${ROOT}/ThirdParty/CyGPUInspector/CyGPUInspectorCore/Include")
target_include_directories(cyinsightovr PRIVATE "${ROOT}/ThirdParty/OpenXR/include")
target_link_libraries(cyinsightovr PRIVATE OvrImGui CyUI::CyUI GLESv3 android log dl)
target_link_options(cyinsightovr PRIVATE "-Wl,-z,max-page-size=16384")
option(CYOVR_EXPERIMENTAL_VULKAN_LAYER "Build the development-only Vulkan discovery probe" OFF)
if(CYOVR_EXPERIMENTAL_VULKAN_LAYER)
    add_library(VkLayer_CyInsightOVR SHARED experimental/VulkanProbe.cpp)
    target_include_directories(VkLayer_CyInsightOVR PRIVATE "${CMAKE_BINARY_DIR}")
    target_link_libraries(VkLayer_CyInsightOVR PRIVATE log)
    target_link_options(VkLayer_CyInsightOVR PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-Bsymbolic")
endif()
